    """Resolve the configured coding-agent repo root only when task start needs it."""

    def resolve(self):
        try:
            repo_root = runtime_repo_root_from_env_or_default(REPO_ROOT)
        except CodingAgentRuntimeConfigError as exc:
            raise CodingTaskError(503, exc.reason_code, exc.summary) from exc
        return GitBaseRefResolver(repo_root=repo_root).resolve()


def resolve_coding_task_service(state: Any | None) -> CodingTaskService:
    """Return the shared CodingTaskService, creating the configured default when absent."""

    service = getattr(state, "coding_task_service", None) if state is not None else None
    if service is not None:
        return service

    state_root = os.environ.get("GASS_CODING_TASK_STATE_ROOT")
    store = _build_configured_store(state_root) if state_root else None
    audit_store = None
    if state_root:
        audit_root = os.environ.get("GASS_CODING_TASK_AUDIT_ROOT") or os.path.join(
            os.path.dirname(state_root),
            "coding_task_audit",
        )
        audit_store = _build_configured_audit_store(audit_root)

    service = CodingTaskService(
        store=store,
        audit_store=audit_store,
        base_ref_resolver=DeferredCodingAgentBaseRefResolver(),
    )
    if state is not None:
        state.coding_task_service = service
    return service


def _build_configured_store(state_root: str) -> DurableCodingTaskStore:
    try:
        return DurableCodingTaskStore(state_root)
    except CodingTaskError:
        raise
    except Exception as exc:
        raise CodingTaskError(503, "coding_task_store_unhealthy", "开发任务状态存储初始化失败。") from exc


def _build_configured_audit_store(audit_root: str) -> CodingAuditStore:
    try:
        return CodingAuditStore(audit_root)
    except CodingTaskError:
        raise
    except Exception as exc:
        raise CodingTaskError(503, "coding_task_audit_store_unhealthy", "开发任务审计存储初始化失败。") from exc
